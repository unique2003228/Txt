async function loadVRMFrom(url){try{const gltf=await vrmLoader.loadAsync(url);const vrm=gltf.userData.vrm||null;if(vrm){VRMUtils.rotateVRM0(vrm);try{VRMUtils.removeUnnecessaryVertices(vrm.scene);}catch(e){}try{VRMUtils.combineSkeletons(vrm.scene);}catch(e){}try{VRMUtils.combineMorphs(vrm);}catch(e){}}return vrm;}catch(e){console.warn("VRM load failed:",url,e);return null;}}
function tryLoadVRM(charId){
  if(VRMBank.ready[charId]) return Promise.resolve(VRMBank.ready[charId]);
  if(VRMBank.prom[charId]) return VRMBank.prom[charId];
  const ch = CHARACTERS.find(c => c.id === charId) || CHARACTERS[0];
  const urls = [ch.url];
  const pr = (async () => {
    for(const u of urls){
      const v = await loadVRMFrom(u);
      if(v){ VRMBank.ready[charId] = v; return v; }
    }
    return null;
  })();
  VRMBank.prom[charId] = pr;
  return pr;
}

function buildSky(){
  const geo=new THREE.SphereGeometry(400,64,40);
  skyMat=new THREE.ShaderMaterial({
    side:THREE.BackSide,depthWrite:false,fog:false,
    uniforms:{
      uTime:{value:0},
      uColorTop:{value:new THREE.Color(0x0a1040)},
      uColorMid:{value:new THREE.Color(0x1a2060)},
      uColorBot:{value:new THREE.Color(0x3a2a6a)}
    },
    vertexShader:`
      varying vec3 vPos;
      void main(){
        vPos=position;
        gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0);
      }`,
    fragmentShader:`
      uniform vec3 uColorTop,uColorMid,uColorBot;
      uniform float uTime;
      varying vec3 vPos;
      
      float hash(vec3 p){
        p=fract(p*0.3183099+vec3(0.1,0.2,0.3));
        p*=17.0;
        return fract(p.x*p.y*p.z*(p.x+p.y+p.z));
      }
      
      void main(){
        vec3 dir=normalize(vPos);
        float h=dir.y*0.5+0.5; // 0 at bottom, 1 at top
        
        // Base gradient
        vec3 col=mix(uColorBot,uColorMid,smoothstep(0.0,0.5,h));
        col=mix(col,uColorTop,smoothstep(0.5,1.0,h));
        
        // Procedural Nebula
        vec3 p=dir*4.0;
        float n=0.0;
        n+=hash(floor(p+uTime*0.02))*0.5;
        n+=hash(floor(p*2.0-uTime*0.03))*0.25;
        n+=hash(floor(p*4.0+uTime*0.05))*0.125;
        n=smoothstep(0.4,0.9,n);
        
        vec3 nebulaCol=mix(vec3(0.4,0.1,0.8),vec3(0.1,0.4,0.9),hash(dir*10.0));
        col+=nebulaCol*n*0.35;
        
        // Stars
        float star=0.0;
        vec3 sp=floor(dir*250.0);
        float sHash=hash(sp);
        if(sHash>0.996){
          float twinkle=sin(uTime*3.0+sHash*100.0)*0.5+0.5;
          star=pow((sHash-0.996)/0.004,2.0)*twinkle;
        }
        col+=vec3(star)*2.5;
        
        // Extra bright stars
        vec3 sp2=floor(dir*120.0);
        float sHash2=hash(sp2);
        if(sHash2>0.998){
          float twinkle2=sin(uTime*2.0+sHash2*50.0)*0.5+0.5;
          star=pow((sHash2-0.998)/0.002,2.0)*twinkle2;
        }
        col+=vec3(0.6,0.8,1.0)*star*3.5;
        
        gl_FragColor=vec4(col,1.0);
      }`
  });
  return new THREE.Mesh(geo,skyMat);
}

function buildWater(){
  const geo=new THREE.PlaneGeometry(800,800,200,200);
  geo.rotateX(-Math.PI/2);
  
  const mat=new THREE.ShaderMaterial({
    uniforms:{
      uTime:{value:0},
      uCamPos:{value:new THREE.Vector3()},
      uSkyColor:{value:new THREE.Color(0x3a2a6a)},
      uHorizonColor:{value:new THREE.Color(0x1a2060)},
      uDeepColor:{value:new THREE.Color(0x0a1040)},
      uHighlight:{value:new THREE.Color(0x80a0ff)},
      uFogNear:{value:15},
      uFogFar:{value:180}
    },
    vertexShader:`
      uniform float uTime;
      varying vec3 vWorldPos;
      varying vec3 vNormal;
      varying float vWave;
      
      float wave(vec2 p,float t){
        float w1=sin(p.x*0.15+t*0.8)*0.35;
        float w2=sin(p.y*0.22+t*0.65)*0.30;
        float w3=sin((p.x+p.y)*0.13+t*1.0)*0.42;
        float w4=sin((p.x-p.y)*0.19+t*0.85)*0.22;
        float w5=sin(p.x*0.45+t*1.4)*0.10;
        float w6=sin(p.y*0.4+t*1.25)*0.08;
        return w1+w2+w3+w4+w5+w6;
      }
      
      void main(){
        vec3 p=position;
        float h=wave(p.xz,uTime);
        
        // Lower the water geometry so it doesn't clip the character
        p.y=h-0.6; 
        vWave=h;
        
        float eps=0.6;
        float hx=wave(p.xz+vec2(eps,0.0),uTime);
        float hz=wave(p.xz+vec2(0.0,eps),uTime);
        vNormal=normalize(vec3(-(hx-h)/eps,1.0,-(hz-h)/eps));
        
        vec4 wp=modelMatrix*vec4(p,1.0);
        vWorldPos=wp.xyz;
        gl_Position=projectionMatrix*viewMatrix*wp;
      }`,
    fragmentShader:`
      uniform vec3 uCamPos;
      uniform vec3 uSkyColor,uHorizonColor,uDeepColor,uHighlight;
      uniform float uFogNear,uFogFar,uTime;
      varying vec3 vWorldPos;
      varying vec3 vNormal;
      varying float vWave;
      
      void main(){
        vec3 N=normalize(vNormal);
        vec3 V=normalize(uCamPos-vWorldPos);
        
        // Reflection
        float fres=pow(1.0-max(dot(N,V),0.0),5.0);
        vec3 col=mix(uDeepColor,uHorizonColor,fres);
        
        // Reflected Sky Color
        vec3 R=reflect(-V,N);
        float skyMask=smoothstep(-0.1,0.8,R.y);
        vec3 skyRef=mix(uHorizonColor,uSkyColor,skyMask);
        col=mix(col,skyRef,fres*0.85);
        
        // Foam / Highlights
        float crest=smoothstep(0.5,1.0,vWave);
        col+=uHighlight*crest*0.4;
        
        // Specular
        vec3 H=normalize(vec3(0.0,1.0,0.0)+V);
        float spec=pow(max(dot(N,H),0.0),100.0);
        col+=uHighlight*spec*2.0;
        
        // Sparkles
        vec2 sp=floor(vWorldPos.xz*2.5+vec2(uTime*0.1));
        float rnd=fract(sin(dot(sp,vec2(12.9898,78.233)))*43758.5453);
        float sparkle=smoothstep(0.995,1.0,rnd);
        float tw=sin(uTime*4.0+rnd*15.0)*0.5+0.5;
        col+=uHighlight*sparkle*tw*2.0;
        
        // Fog
        float dist=length(vWorldPos-uCamPos);
        float f=smoothstep(uFogNear,uFogFar,dist);
        col=mix(col,uDeepColor,f);
        
        gl_FragColor=vec4(col,1.0);
      }`
  });
  
  const w=new THREE.Mesh(geo,mat);
  w.position.y=0;w.frustumCulled=false;
  return w;
}

function updateGame(dt){
  lerpTime(dt);
  
  if(skyMat) skyMat.uniforms.uTime.value+=dt;
  
  const wtr=worldGroup&&worldGroup.userData.water;
  if(wtr){
    wtr.material.uniforms.uTime.value+=dt;
    wtr.material.uniforms.uCamPos.value.copy(camera.position);
  }
  
  let jx=joy.dx,jy=joy.dy;
  if(keys.KeyW||keys.ArrowUp)jy+=1;if(keys.KeyS||keys.ArrowDown)jy-=1;
  if(keys.KeyA||keys.ArrowLeft)jx-=1;if(keys.KeyD||keys.ArrowRight)jx+=1;
  const mag=clamp(Math.hypot(jx,jy),0,1);
  let mx=0,mz=0;
  if(mag>.12){
    const l=Math.hypot(jx,jy);jx/=l;jy/=l;
    const fx=-Math.sin(camYaw),fz=-Math.cos(camYaw),rx=Math.cos(camYaw),rz=-Math.sin(camYaw);
    mx=rx*jx+fx*jy;mz=rz*jx+fz*jy;
    const ll=Math.hypot(mx,mz)||1;mx/=ll;mz/=ll;
    player.yaw=aLerp(player.yaw,Math.atan2(mx,mz),1-Math.exp(-dt*12));}
  const spd=4.2*mag;
  player.pos.x+=mx*spd*dt;player.pos.z+=mz*spd*dt;
  const rr0=Math.hypot(player.pos.x,player.pos.z);
  if(rr0>85){player.pos.x*=85/rr0;player.pos.z*=85/rr0;}
  player.vy-=20*dt;player.pos.y+=player.vy*dt;
  
  const gh=groundAt(player.pos.x,player.pos.z,player.pos.y);
  
  if(player.pos.y <= gh + 0.1 && player.vy <= 0){
    if(!player.onGround && player.vy < -3.5){AudioSys.land();if(localMgr)localMgr.setExpression('surprised',1.0);}
    if (Math.abs(player.pos.y - gh) < 0.3) {player.pos.y = lerp(player.pos.y, gh, 0.4);}
    else {player.pos.y = gh;}
    player.vy = 0; player.onGround = true; player.lastGround = performance.now()/1000;
  } else player.onGround = false;
  
  if(localMgr)localMgr.setOnGround(player.onGround);
  Avatars.forEach(a=>{if(a.mgr.setOnGround)a.mgr.setOnGround(a.mode!=='jump');});
  const nowT=performance.now()/1000;
  if(jumpQueued&&(player.onGround||nowT-player.lastGround<.12)&&nowT-jumpQAt<.2){
    player.vy=7.6;player.onGround=false;jumpQueued=false;AudioSys.jump();
    if(localMgr)localMgr.setExpression('happy',1.2);}
  player.bt+=dt;
  if(!player.onGround)player.mode='jump';
  else if(mag>.12){player.mode='walk';player.wt+=dt*(spd/4.2)*9.5;
    const s=Math.sin(player.wt);
    if(player._ps>0&&s<=0)AudioSys.step();player._ps=s;}
  else player.mode='idle';
  if(localMgr){
    localMgr.root.position.copy(player.pos);
    localMgr.root.rotation.y=player.yaw;
    localMgr.setMode(player.mode,mag);
    localMgr.update(dt);}
  const k=1-Math.exp(-dt*10);
  Avatars.forEach(a=>{
    const o=a.mgr.root;
    if(a.first&&a.target.y>-5){o.position.copy(a.target);o.rotation.y=a.tYaw;a.prev.copy(a.target);a.first=false;}
    else o.position.lerp(a.target,k);
    if(!a.first)o.rotation.y=aLerp(o.rotation.y,a.tYaw,k);
    const spd2=o.position.distanceTo(a.prev)/Math.max(dt,.001);a.prev.copy(o.position);
    a.mgr.setMode(a.mode,clamp(spd2/4.2,.3,1.4));
    a.mgr.update(dt);});
  updateBubbles(dt);
  
  if(sparkPts)sparkPts.rotation.y+=dt*.005;
  
  const off=new THREE.Vector3(Math.sin(camYaw)*Math.cos(camPitch),Math.sin(camPitch),Math.cos(camYaw)*Math.cos(camPitch)).multiplyScalar(camDist);
  camGoal.copy(player.pos).add(new THREE.Vector3(0,.95,0));
  camera.position.lerp(camGoal.clone().add(off),1-Math.exp(-dt*7));
  camera.lookAt(camGoal);
  sun.position.copy(player.pos).add(SHADOW_OFF);
  sunTarget.position.copy(player.pos);sunTarget.updateMatrixWorld();}
